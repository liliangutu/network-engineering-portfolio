! =====================================================
! Device: SW1
! Role: Access Switch
! Project: Enterprise VLAN Segmentation & Inter-VLAN Routing
! Platform: Cisco 2960 / Packet Tracer
! Credentials intentionally redacted
! =====================================================

hostname SW1

spanning-tree mode pvst

! -----------------------------------------------------
! VLANs
! -----------------------------------------------------

vlan 10
 name MANAGEMENT

vlan 20
 name USERS

vlan 30
 name SERVERS

vlan 40
 name GUEST

vlan 99
 name NATIVE


! -----------------------------------------------------
! Access Ports
! -----------------------------------------------------

interface FastEthernet0/1
 description PC1-USERS
 switchport access vlan 20
 switchport mode access
 spanning-tree portfast

interface FastEthernet0/2
 description PC2-GUEST
 switchport access vlan 40
 switchport mode access
 spanning-tree portfast

interface FastEthernet0/3
 description ADMIN-PC-MANAGEMENT
 switchport access vlan 10
 switchport mode access
 spanning-tree portfast


! -----------------------------------------------------
! 802.1Q Trunk to CORE1
! -----------------------------------------------------

interface GigabitEthernet0/1
 description TRUNK_TO_CORE1
 switchport trunk native vlan 99
 switchport trunk allowed vlan 10,20,30,40,99
 switchport mode trunk


! -----------------------------------------------------
! Management Interface
! -----------------------------------------------------

interface Vlan10
 description MANAGEMENT
 ip address 10.10.10.11 255.255.255.0

ip default-gateway 10.10.10.1


! -----------------------------------------------------
! Secure Management
! -----------------------------------------------------

ip domain-name lab.local
ip ssh version 2

username netadmin privilege 15 secret <REDACTED>

ip access-list standard SSH-MANAGEMENT
 permit 10.10.10.0 0.0.0.255
 deny any

line vty 0 15
 access-class SSH-MANAGEMENT in
 login local
 transport input ssh
 exec-timeout 10 0

end
